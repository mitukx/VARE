"""Repeat grouped-query attention key/value heads.

``repeat_kv_heads(keys, n_rep)`` takes one vector per key/value head and
an integer ``n_rep >= 1``. ``keys`` is a non-empty list of equal-length
rows of finite floats.

The result is a new list of length ``len(keys) * n_rep``. Head ``i`` is
repeated ``n_rep`` times in place, before head ``i + 1``:

    [head_0, head_0, ..., head_1, head_1, ...]

with ``n_rep`` copies of each head. Rows are copied. ``keys`` is not
mutated. ``n_rep < 1``, an empty head list, a jagged row, or a non-finite
entry raises ``ValueError``.
"""

from __future__ import annotations

import math


def _validate(keys, n_rep):
    if type(n_rep) is not int or n_rep < 1:
        raise ValueError("n_rep must be an integer >= 1")
    if not keys or any(not row for row in keys):
        raise ValueError("keys must be non-empty")
    width = len(keys[0])
    if any(len(row) != width for row in keys):
        raise ValueError("jagged keys")
    for row in keys:
        for value in row:
            if (
                isinstance(value, bool)
                or not isinstance(value, (int, float))
                or not math.isfinite(value)
            ):
                raise ValueError("keys must be finite")


def repeat_kv_heads(keys, n_rep):
    _validate(keys, n_rep)
    return [row[:] for row in keys for _ in range(n_rep)]
